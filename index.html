RESERVAS DEL ESPACIO ---------- create table if not exists public.reservas ( id uuid primary key default gen_random_uuid(), grupo_id uuid not null references public.grupos(id) on delete cascade, grupo_nombre text not null default '', inicio timestamptz not null, fin timestamptz not null, motivo text not null check (char_length(motivo) between 3 and 200), responsable text not null check (char_length(responsable) between 2 and 100), encargado text not null check (char_length(encargado) between 2 and 100), estado text not null default 'confirmada' check (estado in ('confirmada','cancelada')), creada_en timestamptz not null default now(), check (fin > inicio and fin <= inicio + interval '12 hours'), constraint sin_solapes exclude using gist (tstzrange(inicio, fin) with &&) where (estado <> 'cancelada') ); CHAT: conversaciones privadas y salas ---------- create table if not exists public.salas ( id uuid primary key default gen_random_uuid(), nombre text check (nombre is null or char_length(nombre) between 2 and 60), tipo text not null check (tipo in ('directo','grupal')), creada_por uuid references public.grupos(id) on delete set null, creada_en timestamptz not null default now() ); $$; create function public.iniciar_chat(p_otro uuid) returns uuid language plpgsql security definer set search_path = public as $$ declare v uuid; begin if not public.grupo_aprobado() then raise exception 'grupo no aprobado'; return v; end $$; create function public.sincronizar_sala(p_sala uuid) returns void language plpgsql security definer set search_path = public as $$ begin if not public.es_admin() then raise exception 'solo la direccion';
